async function navigate(url) { if (!url.trim()) return; // Clean up the input if (!url.includes('.') || url.includes(' ')) { url = 'https://www.google.com/search?q=' + encodeURIComponent(url); } else if (!/^https?:\/\//i.test(url)) { url = 'https://' + url; } const status = document.getElementById('status'); const btn = document.getElementById('go-btn'); const proxyFrame = document.getElementById('proxy-frame'); status.textContent = 'Fetching via Node...'; btn.textContent = '...'; // Point the iframe to our custom /proxy route proxyFrame.src = `/proxy?url=${encodeURIComponent(url)}`; proxyFrame.onload = () => { btn.textContent = 'GO'; status.textContent = 'Loaded: ' + url; document.getElementById('frame-bar-url').textContent = url; document.getElementById('frame-container').classList.add('shown'); document.getElementById('frame-container').style.display = 'flex'; }; }
